# The A+B+X+Y+Start soft reset: from the field and from a battle back to the title, then a new game
# that still plays.
# make test-cdi TEST_SCRIPT=tests/soft_reset.txt, then scripts\flycast.ps1 240 -Test.

wait 14000
tap S                      # skip the intro
wait 5000
tap S                      # title: PUSH START
wait 4000
tap A                      # NEW
wait 3000
talk 150000                # the Grand Abbott's farewell
shot sr_field
dump 80053994 5            # the room's looping sounds (FF none): the abbey has one

soft_reset
wait16 8007B2E0 FFFF 3 15000   # the title (game mode 3)
wait 4000
shot sr_title_from_field
dump 80053994 5            # all FF: stopped
tap S                      # title: PUSH START
wait 4000
tap A                      # NEW
wait 3000
talk 150000

warp 2 0 0                 # Holy Plain
wait 1500
to_battle
battle_walk 30000
wait 3000
shot sr_battle
soft_reset
wait16 8007B2E0 FFFF 3 15000
wait 4000
shot sr_title_from_battle
peek32 8008C590            # the low halfword is 8008C592, whose bit 0 is a battle going on

tap S
wait 4000
tap A                      # NEW
wait 3000
talk 150000
warp 2 0 0
wait 1500
shot sr_plain_after
to_battle
battle_walk 30000
wait 3000
shot sr_battle_after
log soft reset finished
