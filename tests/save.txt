tap S                      # title: PUSH START
wait 4000
tap A                      # NEW
wait 3000

wait 1000

wait 500
tap A                      # talk
wait 1500
tap A                      # next page
wait 1500
tap A                      # "Will you stay?" Yes
wait 3000

tap A
wait 2500
shot save_2                # NEW DATA in the first slot
tap A
wait 2500
shot save_3                # saved; "Continue save?"

tap A
